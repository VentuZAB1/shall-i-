def X(p, s):
    return Vector((s * p[0], p[1], p[2]))


def frame(z, n):
    z = Vector(z).normalized()
    x = Vector(n)
    x = (x - z * x.dot(z)).normalized()
    return Matrix((x, z.cross(x), z)).transposed()


def surf_hit(B, blks, origin, direction):
    """exact hit point + normal on the bevelled meshes of blks (CelestialDragon / ThunderRex)"""
    import blockkit as bk
    from mathutils.bvhtree import BVHTree
    B._resolve()
    best = None
    d = Vector(direction).normalized()
    for blk in blks:
        bm = bk._box_bm(blk)
        t = BVHTree.FromBMesh(bm)
        bm.free()
        loc, nrm, _, dist = t.ray_cast(Vector(origin), d)
        if loc is not None and (best is None or dist < best[2]):
            best = (loc, nrm.normalized(), dist)
    if best is None:
        raise RuntimeError("surf_hit: no surface from %s" % (tuple(origin),))
    return best[0], best[1]


def seat_plate(B, blks, origin, direction, half, color, bone, proud=0.010, **kw):
    """plate whose local -y face lies on the surface hit from origin along direction, <= proud above it"""
    p, n = surf_hit(B, blks, origin, direction)
    y = -n
    x = Vector((1, 0, 0)) - n * n.x
    x = x.normalized() if x.length > 1e-4 else Vector((0, 0, 1))
    z = x.cross(y).normalized()
    R = Matrix((x, y, z)).transposed()
    kw.setdefault("studs", [])
    return B.add(p + n * (proud - half[1]), half, color, bone, R=R, **kw)


def prof(b, lz):
    t = (lz / b.h.z + 1) / 2
    fx = b.bot[0] + (b.top[0] - b.bot[0]) * t
    fy = b.bot[1] + (b.top[1] - b.bot[1]) * t
    return b.h.x * fx, b.h.y * fy, b.ts[1] * t


def shell(B, b, y0, y1, za, color, pad=0.022, bone=None, rnd=0.34, **kw):
    """saddle shell over body block b: local y range [y0, y1], local z from za up over the top (ThunderRex)"""
    xa, ya, oa = prof(b, za)
    xb, yb, ob = prof(b, b.h.z)
    zb = b.h.z + pad
    lo_a, hi_a = max(y0 + oa, oa - ya - pad), min(y1 + oa, oa + ya + pad)
    lo_b, hi_b = max(y0 + ob, ob - yb - pad), min(y1 + ob, ob + yb + pad)
    cy, hy = (lo_a + hi_a) / 2, (hi_a - lo_a) / 2
    hx = xa + pad
    c = b.world(Vector((0, cy, (za + zb) / 2)))
    return B.add(c, (hx, hy, (zb - za) / 2), color, bone or b.bone, R=b.R,
                 top=((xb + pad) / hx, (hi_b - lo_b) / 2 / hy), top_shift=(0, (lo_b + hi_b) / 2 - cy), rnd=rnd, **kw)


def fierce_eye(B, c, s, bone="Head", yaw=22.0, tilt=20.0, sc=1.0):
    """long ice-white slit (inner corner low, 20 deg) in a black socket, black slit pupil, heavy DARK brow block; no
    glint, no round pupil"""

    n = R @ Vector((0, -1, 0))
    c = Vector(c)
    B.add(c, (0.062 * sc, 0.008, 0.024 * sc), BLACK, bone, R=R, studs=[], role="detail", bevel=0.004)
    B.add(c + n * 0.006, (0.054 * sc, 0.006, 0.0125 * sc), EYE_HOT, bone, R=R, studs=[], role="detail", bevel=0.003)
    B.add(c + n * 0.011 + R @ Vector((-s * 0.006 * sc, 0, 0)), (0.0055 * sc, 0.004, 0.0125 * sc), BLACK, bone, R=R,
          studs=[], role="detail", bevel=0.002)                                                   # slit pupil
    Rb = Euler((0, math.radians(-s * 22), math.radians(s * yaw)), "XYZ").to_matrix()
    B.add(c + Vector((0, 0, 0.034 * sc)) + n * 0.018 + Rb @ Vector((s * 0.008, 0, 0)), (0.066 * sc, 0.028, 0.017 * sc),
          DARK, bone, R=Rb, role="detail", bevel=0.008)                                          # heavy brow


def frost_fin(B, base, h, chord, thick, bone, rake=0.62):
    """a FLAT crystal fin in the y-z plane (thin across x): hard-edged ICE_W pentagon blade swept back (rake = tip
    offset back / height), an ICE_BELLY inner facet on both faces, a DARK socket at the root. Returns the tip."""
    import blockkit as bk
    b = Vector(base)
    y0, z0 = b.y, b.z - 0.05
    tip = Vector((0, y0 + rake * h, z0 + h + 0.05))
    pts = [Vector((0, y0 - 0.5 * chord, z0)), Vector((0, y0 - 0.30 * chord + 0.18 * h, z0 + 0.50 * h)), tip,
           Vector((0, y0 + 0.5 * chord + 0.22 * h, z0 + 0.42 * h)), Vector((0, y0 + 0.5 * chord, z0))]
    B.poly(pts, thick, ICE_W, bone)
    for sx in (1, -1):                                       # inner facet: a smaller ice-belly blade on each face
        c = sum(pts, Vector()) / len(pts)
        q = [c.lerp(p, 0.62) + Vector((sx * (thick / 2 + 0.004), 0, 0.02)) for p in pts]
        B.poly(q, 0.008, ICE_BELLY, bone)
    B.add(Vector((0, y0, b.z - 0.01)), (thick * 0.95, chord * 0.48, 0.05), DARK, bone, top=(0.8, 0.8),
          role="limb", nostud=True, studs=[])
    return tip


def horn(B, P, w0, bone):
    """3-segment back-swept GOLD horn with GOLD_D ridge bands and a GOLD_D point"""
    for i in range(len(P) - 2):
        a, b = P[i], P[i + 1]
        w = w0 * (1.0 - 0.18 * i)
        B.seg(a, b, w, w * 1.05, GOLD, bone, side=(1, 0, 0), ext=0.05 if i == 0 else 0.02, top=(0.84, 0.84),
              role="limb", nostud=True, studs=[], rnd=0.35)
        d = (b - a).normalized()
        B.seg(b - d * 0.018, b + d * 0.018, w * 0.90, w * 0.95, GOLD_D, bone, side=(1, 0, 0), role="limb",
              nostud=True, studs=[], rnd=0.3)                                                     # ridge band
    a, b = P[-2], P[-1]
    d = (b - a).normalized()
    B.spike(a - d * 0.02, d, (b - a).length + 0.02, w0 * 0.62, GOLD_D, bone, top=(0.12, 0.12), nostud=True)
    return b

        B.seg(W, f, w, w + 0.004, DARK, tb, side=(0, 0, 1), top=(0.38, 0.38), role="limb")

    B.poly([W, f2, e23, f3], 0.028, MEMBRANE, tb)
    B.poly([W, f3, e34, f4], 0.028, MEMBRANE, tb)

        ANCH["FX_FinSpark_" + nm[-1]] = tuple(frost_fin(B, base, h, ch, 0.085, nm))
    # dark dorsal stripe (the back line): a DARK band along the neck back, the body top and the tail top
    for a, b, bone, off, hw in ((N0 + Vector((0, 0.03, -0.05)), N1, "Neck", 0.215, 0.085), (N1, N2, "Neck2", 0.19, 0.07)):
        d = (b - a).normalized()
        bn = Vector((0, -d.z, d.y)) if d.z > 0 else Vector((0, d.z, -d.y))
        if bn.y < 0:
            bn = -bn
        B.seg(a + bn * (off - 0.03), b + bn * (off - 0.03), hw, 0.042, DARK, bone, ext=0.03, ext1=0.0, rnd=0.4)
    prev = None
    for y in (-0.62, -0.42, -0.20, 0.0, 0.20, 0.40, 0.60):
        p, n = surf_hit(B, tops, (0, y, 3.0), (0, 0, -1))

            else:

                      top=(0.75, 0.62), ext1=0.03, rnd=0.36)                                     # thigh
                B.seg(K, A, 0.078, 0.088, STEEL_BLUE, shin, top=(0.80, 0.80), ext=0.035, ext1=0.02)

    T = [Vector(p) for p in TAILP]
    dims = ((0.15, 0.15), (0.115, 0.115), (0.085, 0.087), (0.06, 0.062))

              w * 0.45, 0.02, DARK, bone, ext=0.02 if i else -0.04, top=(0.8, 1), rnd=0.4)        # dorsal band

            dd = (b - a).normalized()
            hgt = d * (1 - 0.18 * t)
            B.spike(p + Vector((0, 0, hgt + 0.01)), (0, 0.6, 0.8) if i < 2 else (0, 0.7, 0.7), 0.09 - 0.015 * i,

    B.add(T[4], (0.07, 0.07, 0.05), DARK, "Tail_Tip", R=frame(tipd, side), role="limb", nostud=True)  # socket

    B.spike(main, tipd + Vector((0, 0, 0.18)), 0.44, 0.085, ICE_W, "Tail_Tip", side=side, top=(0.10, 0.10))
    for sx in (1, -1):

                side=Vector((0, 0, 1)), top=(0.10, 0.10))
    B.spike(main - tipd * 0.02, Vector((0, 0.45, 0.9)), 0.17, 0.04, ICE_BELLY, "Tail_Tip", top=(0.10, 0.10))
    # ================= wings
    for s, side_ in ((1, "L"), (-1, "R")):
        membrane_wing(B, s, side_)
    B.build()
    mn, mx = k.bounds()
    pitch = round(0.2243 * (mx.z - mn.z) / HEIGHT, 4)
    bk.stud_skin(k, pitch)
    return {"TH": round(B.th, 3), "pitch": pitch, "H": round(mx.z - mn.z, 3), "top": round(mx.z, 3),
            "horn_tip": [round(v, 3) for v in ANCH.get("FX_IceTrail_L", (0, 0, 0))],
            "span": round(mx.x - mn.x, 3), "length": round(mx.y - mn.y, 3)}
