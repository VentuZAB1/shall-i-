def _raise(p):
    return _S0 + _ROT @ (Vector(p) - _S0)

def X(p, s):
    return Vector((s * p[0], p[1], p[2]))


def frame(z, n):
    z = Vector(z).normalized()
    x = Vector(n)
    x = (x - z * x.dot(z)).normalized()
    return Matrix((x, z.cross(x), z)).transposed()


def feather(B, root, tip, nrm, width, thick, color, bone, taper=0.22, bars=None):
    """flat tapered feather in the wing plane. bars = [(colour, frac), ...] from the tip backwards: the last fracs of
    the length change colour (e.g. [(CHAR, .16), (FROST_V, .12)] = a violet bar then a charcoal point)."""
    root, tip = Vector(root), Vector(tip)
    R = frame(tip - root, nrm)
    L = (tip - root).length
    d = (tip - root).normalized()
    if not bars:
        return B.add((root + tip) / 2, (thick / 2, width / 2, L / 2), color, bone, R=R, top=(1.0, taper),
                     role="limb", bevel=0.008)
    tot = sum(f for _, f in bars)
    cuts = [(color, 1.0 - tot)] + list(reversed(bars))          # root -> tip
    t0 = 0.0
    w0 = 1.0
    for i, (c, fr) in enumerate(cuts):
        t1 = t0 + fr
        a, b = root + d * (L * t0), root + d * (L * t1)
        w1 = 1.0 - (1.0 - taper) * t1                           # width falls linearly to `taper` at the tip
        last = i == len(cuts) - 1
        ext = 0.006 if i else 0.0
        B.add((a + b) / 2 - d * ext / 2, (thick / 2 + 0.0015 * i, width * w0 / 2, (b - a).length / 2 + ext / 2), c,
              bone, R=R, top=(1.0, (w1 / w0) if not last else taper / w0), role="limb", bevel=0.006 if i else 0.008,
              studs=[], nostud=i > 0)
        t0, w0 = t1, w1
    return tip


PBARS = [(CHAR, 0.11), (FROST_V, 0.17)]          # primaries: a frost-violet bar, then a charcoal point


def wing(B, s, side):
    """3-row wing: SNOW shoulder (marginal coverts) -> STORM_GREY arm bar, STORM_GREY coverts over STORM_GREY
    secondaries with a CHAR trailing edge; CHAR primaries fanned from the wrist, each with a FROST_V bar + CHAR point."""
    S0, W, H, bb = X(WING.S, s), X(WING.W, s), X(H2, s), X(WING.BB, s)
    T = X(TIP, s)
    wb, tb = "Wing_" + side, "WingTip_" + side
    nrm = (T - W).cross(H - W).normalized()
    if nrm.z < 0:
        nrm = -nrm

    def pl(p):
        p = Vector(p)
        return p - nrm * (p - W).dot(nrm)

    S, H, bb = pl(S0), pl(H), pl(bb)
    Wm = S0.lerp(W, 0.40)
    Wa = S0.lerp(W, 0.24)
    B.seg(S0, Wa, 0.065, 0.095, SNOW, wb, side=nrm, ext=0.05, role="limb", top=(0.98, 0.98))
    B.seg(Wa, Wm, 0.064, 0.093, STORM_GREY, wb, side=nrm, ext=0.005, role="limb", top=(0.97, 0.97))
    B.seg(Wm, W, 0.061, 0.09, STORM_GREY, wb, side=nrm, ext=0.01, role="limb", top=(0.85, 0.8))
    B.add(W, (0.055, 0.07, 0.07), STORM_GREY, wb, R=frame(W - S, nrm), role="limb")
    B.seg(W + nrm * 0.08, W.lerp(T, 0.28) + nrm * 0.08, 0.028, 0.05, CHAR, tb, side=nrm, role="limb",
          top=(0.7, 0.7))
    n_sec = 8
    for i in range(n_sec):                                       # secondaries: grey with a charcoal trailing edge
        t = (i + 0.5) / n_sec
        a = S.lerp(W, 0.04 + 0.92 * t)
        end = bb.lerp(H, t)
        d = end - a
        end = end + d.normalized() * (0.07 + 0.04 * ((i % 2) * 2 - 1))
        feather(B, a - nrm * 0.016, end - nrm * 0.016, nrm, 0.27, 0.034, STORM_GREY, wb, taper=0.35,
                bars=[(CHAR, 0.18)])
    for i in range(7):                                           # coverts: plain grey row over the secondaries
        t = (i + 0.5) / 7
        a = S.lerp(W, 0.03 + 0.94 * t)
        end = a.lerp(bb.lerp(H, t), 0.50)
        feather(B, a + nrm * 0.02, end + nrm * 0.02, nrm, 0.22, 0.034, STORM_GREY if t > 0.45 else SNOW, wb,
                taper=0.5, bars=[(FROST_V, 0.24)] if t > 0.22 else None)
    for i, (t, f) in enumerate(((0.12, 0.66), (0.32, 0.77), (0.50, 0.86))):   # inner primaries (arm bone)
        dv = (H - W).normalized().lerp((T - W).normalized(), t).normalized()
        ln = (T - W).length * f
        feather(B, W + dv * 0.04 - nrm * 0.016, W + dv * ln - nrm * 0.016, nrm, 0.25, 0.034, CHAR, wb, bars=PBARS)
    n_pri = 5
    for i in range(n_pri):                                       # outer primaries (tip bone)
        t = i / (n_pri - 1)
        dv = (H - W).normalized().lerp((T - W).normalized(), 0.66 + 0.34 * t).normalized()
        ln = (T - W).length * (0.92 + 0.10 * t) * (1.0 + 0.04 * ((i % 2) * 2 - 1))
        o = nrm * (0.065 + 0.004 * (n_pri - 1 - i))
        tipp = feather(B, W + dv * 0.04 + o, W + dv * ln + o, nrm, 0.23, 0.034, CHAR, tb, bars=PBARS)
        if i == 2:
            ANCH["FX_FrostTrail_" + side] = tuple(tipp)
    for i in range(4):                                           # hand coverts (grey)
        t = (i + 0.5) / 4
        dv = (H - W).normalized().lerp((T - W).normalized(), 0.60 + 0.40 * t).normalized()
        o = nrm * 0.11
        feather(B, W + dv * 0.03 + o, W + dv * 0.38 + o, nrm, 0.20, 0.034, STORM_GREY, tb, taper=0.5)


def eye(B, c, s, yaw=26, bone="Head", sc=1.0):
    """BLACK lid-heavy socket, FROST_V slit iris, a small FROST_H hot spot on the outer half and a BLACK slit pupil
    (no glint, no round pupil); tilted 18 deg, inner corner low"""
    R = Euler((0, math.radians(-s * 18), math.radians(s * yaw)), "XYZ").to_matrix()
    c = Vector(c)
    B.add(c + R @ Vector((s * 0.002, 0, 0.003 * sc)), (0.050 * sc, 0.008, 0.019 * sc), BLACK, bone, R=R, studs=[],
          role="detail", bevel=0.004)
    B.add(c + R @ Vector((0.002 * s, -0.006, -0.002)), (0.045 * sc, 0.006, 0.0125 * sc), FROST_V, bone, R=R,
          studs=[], role="detail", bevel=0.003)
    B.add(c + R @ Vector((s * 0.020 * sc, -0.0095, -0.001)), (0.014 * sc, 0.004, 0.006 * sc), FROST_H, bone, R=R,
          studs=[], role="detail", bevel=0.002)
    B.add(c + R @ Vector((-s * 0.004 * sc, -0.0105, -0.001)), (0.005 * sc, 0.004, 0.0118 * sc), BLACK, bone, R=R,
          studs=[], role="detail", bevel=0.002)                                                # slit pupil


def face_shingles(B, blk, face, rows, color_of, bone, size=(0.055, 0.02, 0.07), tip=None):
    """down-pointing shingles on the -y (face=-1, breast) or +y (face=+1, back) face of a rotated tapered block;
    tip = a colour for a pointed lower tip (barred feathers)"""
    h = blk.h
    tilt = math.degrees(math.atan(h.y * (blk.top[1] - blk.bot[1]) / (2 * h.z)))
    Rf = blk.R @ Euler((math.radians(-face * tilt), 0, 0), "XYZ").to_matrix()
    for r, (lz, xs) in enumerate(rows):
        t = (lz / h.z + 1) / 2
        fy = blk.bot[1] + (blk.top[1] - blk.bot[1]) * t
        fx = blk.bot[0] + (blk.top[0] - blk.bot[0]) * t
        for x in xs:
            p = blk.world(Vector((x * fx, face * (h.y * fy + 0.012), lz)))
            B.add(p, size, color_of(r, x), bone, R=Rf, bot=(0.2, 1.0), role="detail", bevel=0.008)
            if tip:
                q = p + Rf @ Vector((0, face * 0.004, -size[2] * 0.62))
                B.add(q, (size[0] * 0.52, size[1] * 0.75, size[2] * 0.40), tip, bone, R=Rf, bot=(0.15, 1.0),
                      top=(0.75, 1.0), role="detail", bevel=0.006)


def face_bar(B, blk, face, lz, half_w, thick, color, bone, vee=10.0, proud=0.004):
    """a flush chevron bar (two halves meeting in a shallow V pointing DOWN) lying on the -y / +y face of a rotated
    tapered block at local height lz: the snowy-owl breast bar"""
    h = blk.h
    tilt = math.degrees(math.atan(h.y * (blk.top[1] - blk.bot[1]) / (2 * h.z)))
    t = (lz / h.z + 1) / 2
    fy = blk.bot[1] + (blk.top[1] - blk.bot[1]) * t
    fx = blk.bot[0] + (blk.top[0] - blk.bot[0]) * t
    hw = half_w * fx
    for sg in (1, -1):
        Rf = blk.R @ Euler((math.radians(-face * tilt), 0, 0), "XYZ").to_matrix() @ \
            Euler((0, math.radians(sg * vee), 0), "XYZ").to_matrix()
        p = blk.world(Vector((sg * hw / 2, face * (h.y * fy + proud), lz + math.tan(math.radians(vee)) * hw / 2)))
        B.add(p, (hw / 2 + 0.01, 0.008, thick / 2), color, bone, R=Rf, role="detail", bevel=0.004, studs=[])


def neck_cape(B, a, b, r, rows, bone, col=STORM_GREY, size=(0.06, 0.02, 0.09)):
    """flat-laid hackle cape: down-pointing shingles on the nape side of a neck segment"""
    a, b = Vector(a), Vector(b)
    dn = (b - a).normalized()
    xa = Vector((1, 0, 0))
    nn = (Vector((0, 1, 0)) - dn * dn.y).normalized()
    Rn = Matrix.Rotation(math.radians(-14), 3, xa) @ Matrix((xa, nn, dn)).transposed()
    for t, xs in rows:
        c = a.lerp(b, t)
        for x in xs:
            p = c + xa * x + nn * max(0.03, math.sqrt(max(0.0, r * r - x * x)) - 0.004)
            B.add(p, size, col, bone, R=Rn, bot=(0.2, 1.0), role="limb", bevel=0.008)


def plume(B, root, d, L, w, bone, cols=(SNOW, FROST_V), fracs=(0.40, 0.60)):
    """a flat crest plume in the plane facing sideways-ish: SNOW blade, FROST_V pointed tip"""
    d = Vector(d).normalized()
    nrm = Vector((1, 0, 0)) - d * d.x
    nrm.normalize()
    cur = Vector(root) - d * 0.03
    wi = w
    for i, (c, fr) in enumerate(zip(cols, fracs)):
        last = i == len(cols) - 1
        end = cur + d * (L * fr + (0.03 if i == 0 else 0.0))
        w1 = wi * (0.72 if not last else 0.12)
        B.add((cur + end) / 2, (wi * 0.30 + 0.002 * i, wi / 2, (end - cur).length / 2 + 0.005), c, bone,
              R=frame(end - cur, nrm), top=(0.9, w1 / wi), role="limb", bevel=0.005, studs=[], nostud=True)
        cur = end - d * 0.006
        wi = w1
    return cur + d * 0.006
