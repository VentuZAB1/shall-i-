def model(k):
    """STYLE_BIBLE Raiju: a proud thunder wolf, chest forward, big head (HW = TW), short thick legs with navy socks,
    fierce yellow wedge eyes with an iris + highlight, open grin, big white cheek ruffs, chunky bolt inlays on the
    flanks and a thick 3-zig bolt tail. No glow (sparks = particle VFX in Studio)."""
    import blockkit as bk
    B = bk.Blocks(k, seed=11, tier=3)
    # ---- one torso: chest 10 % bigger than the hips, overlapping them by 40 % of its length
    chest = B.add((0, -0.28, 0.84), (0.40, 0.42, 0.35), BLUE, "Chest", bot=(0.90, 0.95))
    hips = B.add((0, 0.22, 0.80), (0.37, 0.42, 0.31), BLUE, "Spine", bot=(0.90, 0.95))
    # white bib with 3 fur points + white belly band (light zone 15-25 %)
    B.on_face(chest, "-y", 0, -0.04, 0.27, 0.25, 0.035, WHITE)
    for x, l in ((-0.15, 0.15), (0.0, 0.19), (0.15, 0.15)):
        B.spike((x, -0.715, 0.62), (0, -0.25, -1), l, 0.07, WHITE, "Chest", top=(0.15, 0.35))
    B.add((0, -0.10, 0.505), (0.26, 0.50, 0.05), WHITE, "Chest", studs=[], role="limb")
    # neck mane: 3 blue-dark tapered tufts on the back of the neck
    for y, z, sz in ((-0.46, 1.20, 1.0), (-0.32, 1.17, 0.85), (-0.18, 1.15, 0.7)):
        B.add((0, y, z), (0.08 * sz, 0.09 * sz, 0.13 * sz), BLUE_D, "Neck", rot=(-40, 0, 0), top=(0.2, 0.3), studs=[])
    # neck block (mostly hidden, joins chest and head)
    B.add((0, -0.56, 1.06), (0.28, 0.22, 0.24), BLUE, "Neck", rot=(28, 0, 0), studs=[])
    # ---- head: HW 0.80 (= TW), HH 0.60, soft top taper
    head = B.add((0, -0.80, 1.36), (0.40, 0.33, 0.30), BLUE, "Head", top=(0.94, 0.94))
    # muzzle: white, 0.60 HW wide, protrudes 0.32, buried 36 %
    muz = B.add((0, -1.20, 1.17), (0.24, 0.25, 0.135), WHITE, "Head", top=(0.88, 0.85), role="limb")
    B.on_face(muz, "+z", 0, 0.06, 0.10, 0.16, 0.02, BLUE)                                   # blue blaze up the nose
    B.add((0, -1.452, 1.255), (0.096, 0.048, 0.057), NAVY, "Head", studs=[], role="limb")    # big nose
    B.add((-0.045, -1.50, 1.285), (0.014, 0.004, 0.014), (246, 244, 250), "Head", studs=[], bevel=0.004)  # sheen
    # open grin: dark mouth plate across the muzzle front, tongue, 2 fangs
    B.add((0, -1.452, 1.075), (0.17, 0.006, 0.038), MOUTH, "Head", studs=[], bevel=0.008)
    B.add((0, -1.456, 1.058), (0.07, 0.006, 0.02), TONGUE, "Head", studs=[], bevel=0.006)
    for s in (1, -1):
        B.spike((s * 0.115, -1.462, 1.11), (0, 0, -1), 0.075, 0.022, (246, 244, 250), "Head")
    # lower jaw (opens in the roar): white
    B.add((0, -1.14, 1.005), (0.20, 0.24, 0.055), WHITE, "Jaw", bot=(0.85, 0.9), role="limb")
    B.add((0, -1.14, 1.062), (0.17, 0.20, 0.006), MOUTH, "Jaw", studs=[], bevel=0.004)
    # fierce eyes (iris + slit + highlight) and 18 deg navy brows
    bk.STYLE["eyes"]["fierce"]["brow"] = (0.30, 0.06, 18)
    B.face_eyes(head, "fierce", IRIS, base=BLUE, brow=NAVY)
    bk.STYLE["eyes"]["fierce"]["brow"] = (0.30, 0.06, 20)
    # big white cheek ruffs: 4 tapered blocks per side
    for s in (1, -1):
        for y, z, dz, l in ((-0.72, 1.30, 0.25, 0.24), (-0.66, 1.18, 0.0, 0.27), (-0.60, 1.08, -0.25, 0.24),
                            (-0.56, 1.22, 0.1, 0.20)):
            B.spike((s * 0.36, y, z), (s * 1.0, 0.55, dz), l, 0.075, WHITE, "Head", top=(0.18, 0.35))
    # ears: tall, thick, yellow inner plate
    for s, side in ((1, "L"), (-1, "R")):
        ear = B.add((s * 0.24, -0.66, 1.86), (0.12, 0.065, 0.24), BLUE, "Ear_" + side, rot=(-8, 0, -s * 12),
                    top=(0.18, 0.5), studs=[], role="limb")
        B.add(ear.world((0, -0.068, -0.03)), (0.078, 0.01, 0.16), YELLOW, "Ear_" + side, R=ear.R, top=(0.15, 1),
              studs=[], bevel=0.008)
    # ---- legs: 1 block + navy sock + paw with 2 toe grooves (front straight, back with a thigh taper)
    for s, side in ((1, "L"), (-1, "R")):
        for fb, y, bot in (("F", -0.42, (1, 1)), ("B", 0.42, (0.88, 0.90))):
            bone = "Leg_" + fb + side
            B.add((s * 0.25, y, 0.40), (0.145, 0.155, 0.31), BLUE, bone, bot=bot)
            B.add((s * 0.25, y, 0.165), (0.135, 0.145, 0.075), NAVY, bone, role="limb", studs=[])
            paw = B.add((s * 0.25, y - 0.03, 0.05), (0.162, 0.186, 0.05), NAVY, bone, role="limb", studs=[])
            for gx in (-0.055, 0.055):
                B.add((s * 0.25 + gx, y - 0.03 - 0.188, 0.05), (0.008, 0.004, 0.036), DARK, bone, studs=[],
                      bevel=0.003)
    # ---- chunky lightning bolt inlays on the chest flanks (0.6 cell wide)
    for s in (1, -1):
        pts = [(-0.58, 1.06), (-0.32, 0.85), (-0.22, 0.93), (0.02, 0.64)]
        P = [Vector((s * 0.402, y, z)) for y, z in pts]
        for a, b in zip(P, P[1:]):
            B.bar(a, b, (s, 0, 0), 0.08, 0.014, YELLOW, "Chest", studs=[], bevel=0.006)
    # ---- tail: furry blue root + a thick 3-zig yellow bolt
    B.add((0, 0.66, 1.02), (0.12, 0.12, 0.15), BLUE, "Tail1", rot=(-50, 0, 0), studs=[], role="limb")
    slab(B, (0, 0.70, 1.06), (0, 0.92, 1.40), 0.16, 0.10, YELLOW, "Tail1", studs=[], role="limb")
    slab(B, (0, 0.95, 1.38), (0, 0.80, 1.50), 0.15, 0.10, YELLOW, "Tail2", studs=[], role="limb")
    B.add((0, 0.94, 1.70), (0.05, 0.085, 0.24), YELLOW, "Tail3", rot=(-34, 0, 0), top=(0.4, 0.12), studs=[],
          role="limb")
    B.build()
    return {"plates": B.stud_count, "TH": round(B.th, 3), "cell": round(B.pcell, 3)}
