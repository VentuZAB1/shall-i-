def X(p, s):
    return Vector((s * p[0], p[1], p[2]))


def frame(z, n):
    z = Vector(z).normalized()
    x = Vector(n)
    x = (x - z * x.dot(z)).normalized()
    return Matrix((x, z.cross(x), z)).transposed()


class PWing(wing3.Wing):
    """wing3.Wing with a pair prefix: bones Wing_<P>_<side> (arm + inner rows) and WingTip_<P>_<side> (hand)"""

    def __init__(self, P, S, W, H, BB):
        super().__init__(S, W, H, BB)
        self.P = P

    def key(self, k, f, up, fwd=0.0, twist=0.0, tip=0.0, side=None):
        for sd, sx in (("L", 1), ("R", -1)):
            if side and sd != side:
                continue
            extra = Quaternion((0, 0, 1), math.radians(-sx * fwd)) @ Quaternion(self.arm_axis(sd), math.radians(sx * twist))
            cm.key_axis(k, "Wing_%s_%s" % (self.P, sd), f, self.root_axis(sd), -sx * up, extra=extra)
            cm.key_axis(k, "WingTip_%s_%s" % (self.P, sd), f, self.hinge_axis(sd), sx * tip)

    def joints(self):
        out = []
        for s, sd in ((1, "L"), (-1, "R")):
            W, H = X(self.W, s), X(self.H, s)
            out += [(W, H, 0.10, "Wing_%s_%s" % (self.P, sd)), (W, W, 0.18, "Wing_%s_%s" % (self.P, sd))]
        return out


def make_wing(P, S0, BB0, W, H2, T, elev):
    S0, BB0 = Vector(S0), Vector(BB0)
    R = Matrix.Rotation(math.radians(-elev), 3, (BB0 - S0).normalized())
    rz = lambda p: S0 + R @ (Vector(p) - S0)
    W, H2, T = rz(W), rz(H2), rz(T)
    N = (T - W).cross(H2 - W).normalized()
    if N.z < 0:
        N = -N
    return PWing(P, tuple(S0), tuple(W), tuple(W + N * 0.30), tuple(BB0)), tuple(H2), tuple(T)

def feather(B, root, tip, nrm, width, thick, color, bone, taper=0.24, tipc=None, tipf=0.30, studs=False, bev=0.004):
    """flat tapered feather in the wing plane; tipc = a coloured pointed tip over the last tipf of the length.
    bev <= 0.004 = an unbevelled 12-tri blade (dense six-wing budget), > 0.004 = a bevelled one"""
    root, tip = Vector(root), Vector(tip)
    R = frame(tip - root, nrm)
    L = (tip - root).length
    kw = dict(role="limb", bevel=max(bev, 0.007)) if studs else dict(role="limb", bevel=bev, nostud=True, studs=[])
    if not tipc:
        return B.add((root + tip) / 2, (thick / 2, width / 2, L / 2), color, bone, R=R, top=(1.0, taper), **kw)
    m = root.lerp(tip, 1 - tipf)
    w1 = 1.0 - (1.0 - taper) * (1 - tipf)
    B.add((root + m) / 2, (thick / 2, width / 2, (m - root).length / 2 + 0.004), color, bone, R=R, top=(1.0, w1), **kw)
    d = (tip - root).normalized()
    B.add((m + tip) / 2 - d * 0.004, (thick / 2 + 0.0015, width * w1 / 2, (tip - m).length / 2 + 0.004), tipc, bone,
          R=R, top=(1.0, taper / w1), role="limb", bevel=bev, nostud=True, studs=[])
    return tip

def angel_wing(B, P, s, side, rows3):
    """dense feather wing: MOON_IVORY arm bar (studded) + wrist + hand bar; a covert row (WING_LILAC studded on the
    upper pair, BODY_MID on the others); WING_LILAC secondaries hanging back to the hinge line; primaries fanned densely
    from the wrist on the hand bone (WING_IN on the upper pair = the one pale highlight, WING_LILAC on the others);
    every flight feather tipped OBSID"""
    wg, h2, tp = WINGS[P]
    S0, W, H, bb, T = X(wg.S, s), X(wg.W, s), X(h2, s), X(wg.BB, s), X(tp, s)
    wb, tb = "Wing_%s_%s" % (P, side), "WingTip_%s_%s" % (P, side)
    nrm = (T - W).cross(H - W).normalized()
    if nrm.z < 0:
        nrm = -nrm
    pl = lambda p: Vector(p) - nrm * (Vector(p) - W).dot(nrm)
    S, H, bb = pl(S0), pl(H), pl(bb)
    sz, n_cov, n_sec, n_pri, fw, bev = WSPEC[P]
    aw = 0.062 * sz + 0.01
    B.seg(S0, W, aw, aw * 1.4, MOON_IVORY, wb, side=nrm, ext=0.05, role="limb", top=(0.82, 0.8))
    B.add(W, (aw * 0.9, aw * 1.1, aw * 1.1), MOON_IVORY, wb, R=frame(W - S, nrm), role="limb")
    B.seg(W + nrm * 0.03, W.lerp(T, 0.25) + nrm * 0.03, aw * 0.5, aw * 0.75, MOON_IVORY, tb, side=nrm, role="limb",
          top=(0.7, 0.7))
    for i in range(n_sec):                                   # secondaries: arm -> hinge line, OBSID tips
        t = (i + 0.5) / n_sec
        a = S.lerp(W, 0.04 + 0.92 * t)
        end = bb.lerp(H, t)
        end = end + (end - a).normalized() * (0.06 + 0.03 * ((i % 2) * 2 - 1)) * sz
        feather(B, a - nrm * 0.016, end - nrm * 0.016, nrm, fw, 0.03, WING_LILAC, wb,
                taper=0.35, tipc=OBSID, tipf=0.12, bev=bev)
    for i in range(n_cov):                                   # coverts
        t = (i + 0.5) / n_cov
        a = S.lerp(W, 0.03 + 0.94 * t)
        end = a.lerp(bb.lerp(H, t), 0.52)
        feather(B, a + nrm * 0.018, end + nrm * 0.018, nrm, fw * 0.80, 0.032, WING_LILAC if rows3 else BODY_MID, wb,
                taper=0.5, studs=rows3, bev=bev)
    for i in range(n_pri):                                   # primaries (hand): WING_IN, OBSID tips
        t = i / (n_pri - 1)
        dv = (H - W).normalized().lerp((T - W).normalized(), 0.12 + 0.88 * t).normalized()
        ln = (T - W).length * (0.66 + 0.36 * t) * (1.0 + 0.03 * ((i % 2) * 2 - 1))
        o = nrm * (0.03 + 0.004 * (n_pri - 1 - i))
        tipp = feather(B, W + dv * 0.04 + o, W + dv * ln + o, nrm, fw * 0.90, 0.03, WING_LILAC,
                       tb, taper=0.22, tipc=OBSID, tipf=0.15, bev=bev)
        if i == n_pri // 2:
            ANCH["FX_FeatherSparkle_%s_%s" % (P, side)] = tuple(tipp)
    if rows3:
        for i in range(3):                                   # hand coverts
            t = (i + 0.5) / 3
            dv = (H - W).normalized().lerp((T - W).normalized(), 0.40 + 0.60 * t).normalized()
            feather(B, W + dv * 0.03 + nrm * 0.065, W + dv * 0.36 + nrm * 0.065, nrm, 0.20, 0.03, WING_IN, tb,
                    taper=0.5, bev=bev)


def eye(B, c, s, yaw=20, bone="Head", sc=1.0):
    """black socket -> BLOOD slit -> black slit pupil, heavy OBSID brow (20 deg, inner end down); no glint"""
    R = Euler((0, math.radians(-s * 20), math.radians(s * yaw)), "XYZ").to_matrix()
    Rb = Euler((0, math.radians(-s * 20), math.radians(s * yaw)), "XYZ").to_matrix()
    c = Vector(c)
    B.add(c + Rb @ Vector((-s * 0.006, 0.026, 0.046 * sc)), (0.074 * sc, 0.032, 0.026 * sc), OBSID, bone, R=Rb,
          role="detail", bevel=0.01)
    B.add(c, (0.058 * sc, 0.008, 0.024 * sc), BLACK, bone, R=R, studs=[], role="detail", bevel=0.004)
    B.add(c + R @ Vector((0.002 * s, -0.007, 0.0)), (0.048 * sc, 0.006, 0.0125 * sc), BLOOD, bone, R=R, studs=[],
          role="detail", bevel=0.003)
    B.add(c + R @ Vector((-s * 0.004, -0.010, 0.0)), (0.006 * sc, 0.004, 0.012 * sc), BLACK, bone, R=R, studs=[],
          role="detail", bevel=0.002)                                                              # slit pupil


def tuft(B, root, d, L, w, color, bone, thick=None):
    d = Vector(d).normalized()
    if thick is None:
        return B.spike(root, d, L, w, color, bone, top=(0.12, 0.38))
    side = Vector((0, 0, 1)).cross(d)
    if side.length < 1e-3:
        side = Vector((1, 0, 0))
    return B.add(Vector(root) + d * L / 2, (w, thick, L / 2), color, bone, R=frame(d, side.normalized()),
                 top=(0.14, 0.45))


def ring_pts(R, n=32):
    return [Vector((R * math.cos(2 * math.pi * i / n), R * math.sin(2 * math.pi * i / n), 0)) for i in range(n)]


def halo(B):
    """the SOLID eclipse: an OBSID disc (filled), a GOLD rim band on the outer edge, a BLOOD corona ring just inside
    the rim on both faces; built flat in the halo frame, tilted HALO_TILT deg forward-up, on the Halo bone"""
    Rt = Matrix.Rotation(math.radians(HALO_TILT), 3, "X")
    c = HALO_C
    xf = lambda p: c + Rt @ p
    t = HALO_T
    disc = [[xf(p + Vector((0, 0, z))) for p in ring_pts(HALO_R - 0.04)] for z in (-t / 2, t / 2)]
    B.tube(disc, OBSID, "Halo", nostud=True)
    rim_o, rim_i = HALO_R, HALO_R - 0.10                   # gold rim band 0.10 (review 3: 0.085; r5 TO 9+)
    n = 32

    def arc(R, z):
        return [xf(Vector((R * math.cos(2 * math.pi * i / n), R * math.sin(2 * math.pi * i / n), z))) for i in range(n + 1)]
    L = HALO_LIP
    B.tube([(arc(rim_o, -t / 2 - L), arc(rim_i, -t / 2 - L)), (arc(rim_o, t / 2 + L), arc(rim_i, t / 2 + L))],
           GOLD, "Halo", closed=False, nostud=True)
    cor_o = rim_i
    if HALO_BAND > 0:                                        # darker gold inner band on each face (review 1 NICE 3)
        for zs in (1, -1):
            z0, z1 = zs * (t / 2 - 0.002), zs * (t / 2 + 0.008)
            lo, hi = (z0, z1) if zs > 0 else (z1, z0)
            B.tube([(arc(rim_i, lo), arc(rim_i - HALO_BAND, lo)), (arc(rim_i, hi), arc(rim_i - HALO_BAND, hi))],
                   GOLD_D, "Halo", closed=False, nostud=True)
        cor_o = rim_i - HALO_BAND
    for zs in (1, -1):                                       # corona ring on each face
        z0, z1 = zs * (t / 2 - 0.002), zs * (t / 2 + 0.005)
        lo, hi = (z0, z1) if zs > 0 else (z1, z0)
        B.tube([(arc(cor_o, lo), arc(cor_o - 0.12, lo)), (arc(cor_o, hi), arc(cor_o - 0.12, hi))], BLOOD, "Halo",
               closed=False, nostud=True)
    ANCH["FX_CoronaSparkle"] = tuple(xf(Vector((0, -HALO_R, t / 2))))

def mix(a, b, t):

def walk(k):
    """lateral-sequence lion walk (The Cat Returns): the back and head stay level, the wings stay in the display
    fan (the upper pair 4 lower), the halo trails the head by 3 f (a small counter-roll), the tail sways"""
    n = 40

    def body(f):
        b = 0.5 - 0.5 * math.cos(4 * math.pi * f / n)
        k.key("Spine", f, rot=(0.6 * pk.wave(f, n, 0.1, 2), 1.5 * pk.wave(f, n, 0.1), 1.6 * pk.wave(f, n)),
              loc=(0, 0, 0.012 * b))
        k.key("Chest", f, rot=(0.5, 1.0 * pk.wave(f, n, 0.25), -0.6 * pk.wave(f, n)))
        head(k, f, -1.0 * pk.wave(f, n, 0.2, 2), -2.5 * pk.wave(f, n, 0.25))

    return cm.quad_walk(k, n, 0.15, 0.08, body, gait="walk")

        k.key("Spine", f, rot=(0, 0, 0), loc=(0, 0, 0))

def wingcascade(k):
    """SIGNATURE - SIX-WING CASCADE (110 f, in place): f0-14 the Seraph bows its crown (head down 6, the halo tilts
    with it); each pair then lifts with a TIP-TO-ROOT ripple (the hand rises first, the arm follows 6 f later): upper
    pair f14-28 (event `ripple1` f22), middle pair f30-44 (`ripple2` f40), lower pair f44-58 (`ripple3` f54); the six
    wings hold the full fan f58-64; f64-94 the reverse ripple (lower pair folds first, then the middle, then the
    upper); the head lifts, the halo levels; the exact rest pose from f100."""
    n = 110
    k.new_action("act_wingcascade", n, False)
    rest_keys(k, 0)
    sched = {"A": (14, 22, 28, 78, 86, 94), "B": (30, 38, 44, 70, 78, 86), "C": (44, 52, 58, 64, 72, 80)}
    for P, (t0, t1, t2, r0, r1, r2) in sched.items():
        d, fa = DISPLAY_POSE[P], FAN[P]
        mid = {"C": (14.0, d[1], d[2], fa[3]),                # review 5: the lower pair LIFTS, then settles level;
               "B": (d[0] + 4.0, 0.5 * (d[1] + fa[1]), d[2], 0.5 * (d[3] + fa[3]))}.get(P, (d[0], d[1], d[2], fa[3]))
        for f, w in ((0, d), (t0, d), (t1, mid), (t2, fa), (r0, fa),     # B sweeps back as it rises (chest)
                     (r1, (d[0] + (fa[0] - d[0]) * 0.5, fa[1], fa[2], fa[3])), (r2, d), (n, d)):
            wings(k, f, P, *w)
    for f, hp, hz, ch in ((14, 6, -0.02, 0), (28, 6, -0.02, -1.5), (58, 6, -0.02, -2.5), (64, 6, -0.02, -2.5),
                          (90, 0, 0.0, -1), (100, 0, 0.0, 0)):
        k.key("Spine", f, rot=(0, 0, 0), loc=(0, 0, 0))
        k.key("Chest", f, rot=(ch, 0, 0))
        head(k, f, hp)
        k.key("Jaw", f, rot=(0, 0, 0))
        k.key("Mane", f, rot=(0, 0, 0))
        halo_key(k, f, 0, 0, hz)
        tail_keys(k, f, 3 if 28 <= f <= 64 else 0, 0)
    rest_keys(k, n)
    plant_frames(k, range(0, n + 1))
    k.finish(loop=False)
    k.events["act_wingcascade"] = {"ripple1": round(22 / n, 3), "ripple2": round(40 / n, 3),
                                   "ripple3": round(54 / n, 3)}


def anim(k):
    idle(k)
    stride = walk(k)
    fly(k)
    takeoff(k)
    land(k)
    roar(k)
    wingcascade(k)
    return stride


LEGS4 = ["Leg_FL", "Leg_FR", "Leg_BL", "Leg_BR"]
TAIL = ["Tail1", "Tail2", "Tail3", "Tail4"]
WBONES = ["%s_%s_%s" % (b, P, s) for P in "ABC" for s in "LR" for b in ("Wing", "WingTip")]


def check_pairs():
    P = []
    for i, a in enumerate(WBONES):
        for b in WBONES[i + 1:]:
            if a.split("_")[1:] == b.split("_")[1:]:
                continue                                  # arm vs its own hand (hinge) is checked by joints()
            P.append((a, b))
        for b in ["Head", "Neck", "Chest", "Spine", "Mane", "Halo", "Tail1", "Tail2"] + LEGS4:
            P.append((a, b))
    for s in ("L", "R"):
        P += [("Leg_F" + s, "Leg_B" + s), ("Shin_F" + s, "Shin_B" + s), ("Foot_F" + s, "Foot_B" + s)]
    P += [("Shin_FL", "Shin_FR"), ("Shin_BL", "Shin_BR"), ("Foot_FL", "Foot_FR"), ("Foot_BL", "Foot_BR")]
    for a in ("Head", "Jaw", "Mane"):
        for b in ("Chest", "Leg_FL", "Leg_FR", "Shin_FL", "Shin_FR", "Foot_FL", "Foot_FR"):
            P.append((a, b))
    P += [("Halo", "Head"), ("Halo", "Neck"), ("Halo", "Mane"), ("Jaw", "Neck")]
    for t in ("Tail2", "Tail3", "Tail4"):
        for b in ("Leg_BL", "Leg_BR", "Shin_BL", "Shin_BR", "Foot_BL", "Foot_BR"):
            P.append((t, b))
    return P
